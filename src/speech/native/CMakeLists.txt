cmake_minimum_required(VERSION 3.19)
project(unilink_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT WHISPER_SOURCE_DIR)
  message(FATAL_ERROR "Set WHISPER_SOURCE_DIR to the pinned whisper.cpp source")
endif()
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper)
# Keep the 130 MB generated data table out of a single compiler invocation.
find_program(NODE_EXECUTABLE node REQUIRED)
set(shader_table "${CMAKE_CURRENT_BINARY_DIR}/whisper/ggml/src/ggml-vulkan/mul_mm.comp.cpp")
set_source_files_properties("${shader_table}" TARGET_DIRECTORY ggml-vulkan PROPERTIES HEADER_FILE_ONLY TRUE)
set(shader_parts)
foreach(index RANGE 0 15)
  list(APPEND shader_parts "${CMAKE_CURRENT_BINARY_DIR}/shader-parts/mul-mm-${index}.cpp")
endforeach()
add_custom_command(OUTPUT ${shader_parts}
  COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/split-shaders.mjs" "${shader_table}" "${CMAKE_CURRENT_BINARY_DIR}/shader-parts"
  DEPENDS "${shader_table}" "${CMAKE_CURRENT_SOURCE_DIR}/split-shaders.mjs"
  VERBATIM)
add_custom_target(unilink-shader-parts DEPENDS ${shader_parts})
set_source_files_properties(${shader_parts} TARGET_DIRECTORY ggml-vulkan PROPERTIES GENERATED TRUE)
target_sources(ggml-vulkan PRIVATE ${shader_parts})
add_executable(unilink-whisper bridge.cpp)
target_link_libraries(unilink-whisper PRIVATE whisper common json_cpp)
if(MINGW)
  target_link_options(unilink-whisper PRIVATE -static-libgcc -static-libstdc++)
endif()
